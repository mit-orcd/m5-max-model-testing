        echo "Usage: find_dupes <dir>" >&2

    declare -A md5map

    # Gather md5 hashes for all regular files under $dir

        # md5 -r prints: "<hash> <filename>"

    local dir="$1"
    if [[ -z "$dir" || ! -d "$dir" ]]; then
        return 1

    # Associative array to map md5 hash to list of files
    declare -A hash_to_files

    # Find all regular files recursively
    while IFS= read -r -d '' file; do
        # Compute MD5 hash using BSD md5 -r (format: hash  filename)
        # We only need the hash part
        local hash
        hash=$(md5 -r "$file" 2>/dev/null | cut -d' ' -f1)
