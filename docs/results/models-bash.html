function top_errors() {
    local file="$1"
    local n="$2"
    local err_counts=()
    local err_sigs=()

    if [[ ! -f "$file" ]]; then
        echo "File not found: $file"
        return 1
    fi

    while IFS= read -r line; do
        if [[ $line == ERROR:* ]]; then
            local sig="${line##*ERROR:}"
            sig="${sig//[0-9]/#}"
            err_counts+=("$sig")
